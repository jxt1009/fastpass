```swift
import XCTest
import CoreLocation
@testable import FastTrack

final class IncrementalDistanceTests: XCTestCase {

    func test_incrementalDistanceMatchesFullSweep() {
        // Build a synthetic polyline of 10 points 100m apart on a north-bound line.
        // Reference distance computed via the O(n) sweep over the same input.
        var coords: [CLLocationCoordinate2D] = []
        for i in 0..<10 {
            coords.append(CLLocationCoordinate2D(latitude: 37.0 + Double(i) * 0.0009, longitude: -122.0))
        }
        let locs = coords.map { CLLocation(latitude: $0.latitude, longitude: $0.longitude) }
        var reference: Double = 0
        for i in 1..<locs.count {
            reference += locs[i-1].distance(from: locs[i])
        }

        // Incremental form: cumulative = sum of deltas.
        var cumulative: Double = 0
        var last: CLLocation?
        for loc in locs {
            if let last = last {
                cumulative += last.distance(from: loc)
            }
            last = loc
        }
        XCTAssertEqual(cumulative, reference, accuracy: 0.5)
    }

    func test_zeroDistanceForSinglePoint() {
        let loc = CLLocation(latitude: 0, longitude: 0)
        var cumulative: Double = 0
        var last: CLLocation?
        if let last = last {
            cumulative += last.distance(from: loc)
        }
        last = loc
        XCTAssertEqual(cumulative, 0)
    }
}
```

```swift
internal func processLocation(_ location: CLLocation) {
    let age = abs(location.timestamp.timeIntervalSinceNow)
    if age > 5.0 { return }

    let speed = routePointSpeed(for: location)
    let speedMph = speed * 2.23694

    // Incremental distance: one O(1) delta per tick instead of O(n) re-sweep.
    if let prev = lastDistanceTickLocation {
        let delta = prev.distance(from: location)
        // Reject zero/negative deltas (jitter, duplicates) and unreasonable jumps.
        if delta > 0.5 && delta < 500 {
            runningDistanceMeters += delta
        }
    }
    lastDistanceTickLocation = location

    // Single source of truth: richRoutePoints. Drop recordingLocations
    // and routeCoordinates appends (handled in Task 3 / D-3).
    richRoutePoints.append((
        lat: location.coordinate.latitude,
        lng: location.coordinate.longitude,
        speed: speed,
        ts: location.timestamp.timeIntervalSince1970
    ))

    if speed > currentMaxSpeed { currentMaxSpeed = speed }

    Task.detached { [weak self] in
        await self?.processLocationHeavy(location, speed: speed, speedMph: speedMph)
    }

    updateCurrentDrive()
    if publishThrottler.shouldPublish() {
        updateLiveActivity(speedMph: speedMph, distanceMiles: metersToMiles(runningDistanceMeters))
    }
}

private func metersToMiles(_ m: Double) -> Double { m / 1609.344 }
```

```swift
var currentDriveInternal: Drive? {
    get { _currentDrive }

        }
```

```swift
struct RouteSerializerOutput: Sendable {
    let v1String: String  // existing JSON string for backward compat
    let v2Array: [[String: Any]]  // additive, server may ignore

static func encode(_ snapshot: RouteSerializer.Snapshot) -> RouteSerializerOutput {
    let pointDicts: [[String: Any]] = snapshot.richRoutePoints.map { p in
        ["lat": p.lat, "lng": p.lng, "speed": p.speed, "ts": p.ts]

    return RouteSerializerOutput(v1String: v1String, v2Array: pointDicts)
}
```
